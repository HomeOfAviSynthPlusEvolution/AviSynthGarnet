cmake_minimum_required(VERSION 3.24)
project(AviSynthGarnet VERSION 0.0.1 LANGUAGES C CXX)
option(GARNET_BUILD_PLUGIN "Build public AviSynth+ SDK plugin" OFF)
option(GARNET_BUILD_TESTS "Build Garnet tests" ${PROJECT_IS_TOP_LEVEL})
set(GARNET_AVS_SDK_DIR "" CACHE PATH "Explicit directory containing avisynth.h")
set(GARNET_TEST_AVS_RUNTIME "" CACHE FILEPATH "Existing runtime for opt-in frame integration test")
include(cmake/Mruby.cmake)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/runtime.rb")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/runtime.rb" GARNET_RUNTIME)
configure_file(src/runtime.hpp.in generated/runtime.hpp @ONLY)
add_library(garnet_engine STATIC src/engine.cpp)
add_library(Garnet::Engine ALIAS garnet_engine)
find_package(Threads REQUIRED)
target_compile_features(garnet_engine PUBLIC cxx_std_17)
target_include_directories(garnet_engine PUBLIC include)
target_include_directories(garnet_engine PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(garnet_engine PRIVATE garnet_mruby Threads::Threads)
set_target_properties(garnet_engine PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(MSVC)
  # mruby uses C++ exceptions through C-linkage functions internally. /EHc
  # would incorrectly let clang-cl assume those functions cannot unwind.
  target_compile_options(garnet_engine PRIVATE /W4 /permissive- /EHs /EHc-)
else()
  target_compile_options(garnet_engine PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(garnet_engine PRIVATE m)
endif()
if(GARNET_BUILD_PLUGIN)
  if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR "Public Windows C++ plugin requires MSVC or clang-cl, not MinGW")
  endif()
  if(NOT EXISTS "${GARNET_AVS_SDK_DIR}/avisynth.h")
    message(FATAL_ERROR "GARNET_AVS_SDK_DIR must contain the public AviSynth+ SDK")
  endif()
  add_library(garnet_plugin SHARED adapters/avsplus/plugin.cpp)
  target_link_libraries(garnet_plugin PRIVATE Garnet::Engine)
  target_include_directories(garnet_plugin PRIVATE src "${GARNET_AVS_SDK_DIR}")
  set_target_properties(garnet_plugin PROPERTIES PREFIX "" OUTPUT_NAME "Garnet"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endif()
if(GARNET_BUILD_TESTS)
  enable_testing()
  add_executable(garnet_execution_tests tests/execution.cpp)
  target_compile_features(garnet_execution_tests PRIVATE cxx_std_17)
  target_include_directories(garnet_execution_tests PRIVATE src)
  target_link_libraries(garnet_execution_tests PRIVATE garnet_mruby Threads::Threads)
  if(MSVC)
    target_compile_options(garnet_execution_tests PRIVATE /EHs /EHc-)
  endif()
  add_test(NAME garnet_execution COMMAND garnet_execution_tests)
  set_tests_properties(garnet_execution PROPERTIES TIMEOUT 30)
  add_executable(garnet_contract_tests tests/contract.c)
  target_link_libraries(garnet_contract_tests PRIVATE Garnet::Engine)
  set_target_properties(garnet_contract_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
  add_test(NAME garnet_contract COMMAND garnet_contract_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/ruby")
  add_executable(garnet_concurrency_tests tests/concurrency.cpp)
  target_link_libraries(garnet_concurrency_tests PRIVATE Garnet::Engine Threads::Threads)
  add_test(NAME garnet_concurrency COMMAND garnet_concurrency_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/ruby")
  set_tests_properties(garnet_concurrency PROPERTIES TIMEOUT 30 PROCESSORS 4)
  add_executable(garnet_release_tests tests/release.cpp)
  target_link_libraries(garnet_release_tests PRIVATE Garnet::Engine Threads::Threads)
  add_test(NAME garnet_release COMMAND garnet_release_tests)
  set_tests_properties(garnet_release PROPERTIES TIMEOUT 30 PROCESSORS 2)
  if(GARNET_BUILD_PLUGIN AND WIN32 AND GARNET_TEST_AVS_RUNTIME)
    if(NOT EXISTS "${GARNET_TEST_AVS_RUNTIME}")
      message(FATAL_ERROR "GARNET_TEST_AVS_RUNTIME does not exist")
    endif()
    add_executable(garnet_avs_tests tests/avsplus.cpp)
    target_compile_features(garnet_avs_tests PRIVATE cxx_std_17)
    target_include_directories(garnet_avs_tests PRIVATE "${GARNET_AVS_SDK_DIR}")
    add_test(NAME garnet_avs_frames COMMAND garnet_avs_tests
      "${GARNET_TEST_AVS_RUNTIME}" "$<TARGET_FILE:garnet_plugin>" "${CMAKE_CURRENT_SOURCE_DIR}/examples/resize.avs")
    set_tests_properties(garnet_avs_frames PROPERTIES TIMEOUT 30)
    add_test(NAME garnet_avs_libraries COMMAND garnet_avs_tests
      "${GARNET_TEST_AVS_RUNTIME}" "$<TARGET_FILE:garnet_plugin>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/ruby/export.avs")
    set_tests_properties(garnet_avs_libraries PROPERTIES TIMEOUT 30)
    foreach(_pipeline avs-avs avs-ruby ruby-avs ruby-ruby nested failure cycle native-failure native-cycle deferred)
      add_test(NAME garnet_pipeline_${_pipeline} COMMAND garnet_avs_tests
        "${GARNET_TEST_AVS_RUNTIME}" "$<TARGET_FILE:garnet_plugin>"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/ruby/pipeline/${_pipeline}.avs")
      set_tests_properties(garnet_pipeline_${_pipeline} PROPERTIES TIMEOUT 30)
    endforeach()
    add_test(NAME garnet_pipeline_deferred_mt COMMAND garnet_avs_tests
      "${GARNET_TEST_AVS_RUNTIME}" "$<TARGET_FILE:garnet_plugin>"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/ruby/pipeline/deferred-mt.avs" "stamp")
    set_tests_properties(garnet_pipeline_deferred_mt PROPERTIES TIMEOUT 30 PROCESSORS 4)
  endif()
endif()
